# Helios Support Mailbox — runtime dependencies
# Python 3.10+
#
#   pip install -r requirements.txt
#   python server.py          # -> http://0.0.0.0:1222

# --- web framework ---------------------------------------------------------
# ASGI app, request validation, static file serving.
fastapi>=0.110,<1.0

# ASGI server. server.py calls uvicorn.run() directly, no CLI needed.
# [standard] pulls in uvloop + httptools (faster) and watchfiles (--reload).
# Swap to bare `uvicorn` if you hit build issues on Windows.
uvicorn[standard]>=0.27,<1.0

# --- model calls -----------------------------------------------------------
# Async HTTP client used by llm.py to talk to the OpenAI-compatible endpoint.
# Deliberately not using the openai SDK: llm.py needs to strip request params
# the model family rejects and retry, which is easier against the raw API.
httpx>=0.27,<1.0

# --- request/response models ----------------------------------------------
# FastAPI depends on this too; pinned explicitly because server.py imports it.
pydantic>=2.5,<3.0

# --- prompt templating -----------------------------------------------------
# tags.py renders email_automatic_reply_en_US.jinjia2 (the step-1 analyser
# prompt). Kept as a real template file so the intent/sentiment taxonomy can be
# edited without touching Python.
jinja2>=3.1,<4.0

# --- notes -----------------------------------------------------------------
# .env is parsed by server.load_env(), so python-dotenv is NOT required.
# History is stored in sqlite3 from the standard library, no driver needed.
